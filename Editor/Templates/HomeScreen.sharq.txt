<template>
    <ui:VisualElement name="Root" class="home-screen">
<!-- Greeting -->
        <ui:Label name="Title" class="home-screen__title" text="Welcome to SUS" />
        <ui:Label name="Subtitle" class="home-screen__subtitle" text="Vue-like UI framework for Unity" />

<!-- Go to settings screen -->
        <ui:Button name="SettingsBtn" class="home-screen__button"
text="Settings" @click="GoSettings" />

<!-- Counter (demonstration of reactivity) -->
        <ui:VisualElement name="Counter" class="home-screen__counter">
            <ui:Label name="CounterLabel" :text="CountLabel" />
            <ui:Button name="IncBtn" text="+" @click="Increment" />
        </ui:VisualElement>
    </ui:VisualElement>
</template>

<script>
using Sharq.Core;
using Sharq.Router;

// Props
public Prop<int> Count = new(0);

// Computed
public string CountLabel => $"Counter: {Count.Value}";

void GoSettings()
{
// Go through the router
    Router.Value?.Navigate("/settings");
}

void Increment()
{
    Count.Value++;
}
</script>
