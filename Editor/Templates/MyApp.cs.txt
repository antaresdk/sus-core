using Sharq.Core;
using Sharq.Router;
using UnityEngine;
using UnityEngine.UIElements;

/// <summary>
/// The entry point of the SUS application is minimal bootstrap.
/// Assign this script to a GameObject with a free UIDocument.
/// </summary>
public class MyApp : MonoBehaviour
{
    [SerializeField] private UIDocument _uiDocument;

    private void Start()
    {
// 1. Create an application
        // 2. Select a theme (Dark / Light / new SusTheme("midnight"))
        // 3. Mount the router with a route map
        // 4. Launch
        SusApp.Create(_uiDocument)
            .UseTheme(SusTheme.Dark)
            .UseRouter(router =>
            {
                router.AddRoute("/", () => new HomeScreen());
                router.AddRoute("/settings", () => new SettingsScreen());
            })
            .Run();
    }
}
