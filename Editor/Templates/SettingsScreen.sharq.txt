<template>
    <ui:VisualElement name="Root" class="settings-screen">
<!-- Title -->
        <ui:VisualElement name="Header" class="settings-screen__header">
<ui:Button name="BackBtn" text="← Back" @click="GoBack" class="settings-screen__back-btn" />
            <ui:Label name="Title" text="Settings" class="settings-screen__title" />
        </ui:VisualElement>

<!-- Section: Theme -->
        <ui:VisualElement name="ThemeSection" class="settings-screen__section">
<ui:Label name="ThemeLabel" text="Theme" class="settings-screen__section-label" />
            <ui:VisualElement name="ThemeToggle" class="settings-screen__toggle-row">
                <ui:Label name="ThemeToggleLabel" :text="ThemeLabel" class="settings-screen__toggle-label" />
<ui:Button name="ToggleBtn" text="Change Theme" @click="ToggleTheme" class="settings-screen__toggle-btn" />
            </ui:VisualElement>
        </ui:VisualElement>

<!-- Section: Language -->
        <ui:VisualElement name="LangSection" class="settings-screen__section">
<ui:Label name="LangLabel" text="Language" class="settings-screen__section-label" />
            <ui:Label name="LangValue" :text="Language" class="settings-screen__section-value" />
        </ui:VisualElement>
    </ui:VisualElement>
</template>

<script>
using Sharq.Core;
using Sharq.Router;

// Props
public Prop<string> Language = new("Russian");

// Computed
public string ThemeLabel => SusThemeService.Current.Value == SusTheme.Dark
? "Dark"
    : "Light";

void GoBack()
{
    Router.Value?.Navigate("/");
}

void ToggleTheme()
{
    var root = this;
    while (root?.parent != null) root = root.parent;

    var next = SusThemeService.Current.Value == SusTheme.Dark
        ? SusTheme.Light
        : SusTheme.Dark;

    SusThemeService.Instance.SetTheme(root, next);
}
</script>
