<template>
    <ui:VisualElement $MainElement class="#SCRIPTNAME_LOWER#-modal" v-if="Model.Value">
        <ui:VisualElement class="#SCRIPTNAME_LOWER#-modal__backdrop"
                          name="backdrop" @click="OnBackdropClick" />

        <ui:VisualElement class="#SCRIPTNAME_LOWER#-modal__card" name="card">
            <ui:VisualElement class="#SCRIPTNAME_LOWER#-modal__header">
                <ui:Label name="Title" text="#SCRIPTNAME#" />
                <ui:Button name="CloseBtn" text="✕" @click="Close"
                           class="#SCRIPTNAME_LOWER#-modal__close-btn" />
            </ui:VisualElement>

            <ui:VisualElement class="#SCRIPTNAME_LOWER#-modal__body">
                <slot />
            </ui:VisualElement>

            <ui:VisualElement class="#SCRIPTNAME_LOWER#-modal__footer">
                <slot name="footer" />
            </ui:VisualElement>
        </ui:VisualElement>
    </ui:VisualElement>
</template>

<script>
using Sharq.Core;

public Prop<bool> Model = new(false);

public System.Action OnOpened;
public System.Action OnClosed;

public void Open()
{
    Model.Value = true;
    OnOpened?.Invoke();
}

public void Close()
{
    Model.Value = false;
    OnClosed?.Invoke();
}

void OnBackdropClick()
{
    Close();
}

protected override void Created()
{
    Watch(Model, (_, isOpen) =>
    {
        $MainElement.style.display = isOpen ? DisplayStyle.Flex : DisplayStyle.None;
    });
}
</script>
